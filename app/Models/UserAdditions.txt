use Illuminate\Database\Eloquent\SoftDeletes;
use App\Traits\Auditable;

public function role() { return $this->belongsTo(Role::class); }

public function can_do(string $permission): bool
{
    if ($this->role?->name === 'owner') return true;
    return $this->permissionKeys()->contains($permission);
}
public function permissionKeys()
{
    return cache()->remember("user.{$this->id}.perms", 300, fn () =>
        $this->role ? $this->role->permissions()->pluck('key') : collect());
}
